# CN 站点临时关闭片段（粘贴用）
#
# 用法：把下面以 `# >>> sagemro-maintenance >>>` 开头、
# 以 `# <<< sagemro-maintenance <<<` 结尾的三行，
# 分别粘贴到阿里云 ECS 上每一个 443 server 块内（紧跟 `server {` 之后即可）：
#
#   - sagemro.cn / www.sagemro.cn
#   - ai.sagemro.cn
#   - admin.sagemro.cn
#   - engineer.sagemro.cn
#
# 不要粘贴到 api.sagemro.cn 的 server 块：那是反向代理，暂停它请用
# GitHub Actions 里手动 dispatch 的 deploy-maintenance（见 ops/maintenance/README.md）。
#
# 粘贴后执行：sudo nginx -t && sudo systemctl reload nginx
# 删除这三行即恢复，再执行一次同样两条命令。
#
# 注意：`return` 在同一个 server 块里只能出现一次。若 `nginx -t` 报
# "return directive is duplicate"，说明该块本来就有 server 级 return，
# 跳过该块（它通常只是 80→443 的跳转块，本来就不需要在其中加维护页）。

    # >>> sagemro-maintenance >>>
    default_type text/html;
    return 503 '<!doctype html><html lang="zh-CN"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><meta name="robots" content="noindex, nofollow, noarchive"><title>系统维护中 - SAGEMRO</title><style>body{margin:0;min-height:100vh;display:grid;place-items:center;padding:24px;background:#f4f5f7;color:#17191f;font:16px/1.7 system-ui,sans-serif}main{width:100%;max-width:32rem;background:#fff;border:1px solid #e3e6eb;border-radius:14px;padding:32px}h1{margin:0 0 14px;font-size:22px;line-height:1.35}p{margin:0 0 14px;color:#3d424c}p:last-child{margin:0}a{color:#0b4fd1}</style></head><body><main><h1>系统维护中</h1><p>SAGEMRO 网站正在进行计划内维护，暂时无法访问。我们正在尽快恢复服务，数据不受影响。</p><p>需要立即联系：<a href="mailto:support@sagemro.com">support@sagemro.com</a></p></main></body></html>';
